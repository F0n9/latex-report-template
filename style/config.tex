% Tiếng Việt và Mã hóa ký tự
\usepackage[utf8]{vietnam}
\usepackage[utf8]{inputenc}
\usepackage[vietnamese,english]{babel}
\usepackage[T5,T1]{fontenc}

% Phông chữ (Times New Roman tương đương trong LaTeX)
\usepackage{mathptmx}
\usepackage{amsmath}
\usepackage{amssymb}

% Bảng màu (Đồng bộ 100% với Typst theme.typ)
\usepackage[table,xcdraw,svgnames]{xcolor}
\definecolor{titlecolor}{HTML}{1E3778}   % Deep Navy Blue
\definecolor{bodycolor}{HTML}{111111}    % Dark Charcoal Body
\definecolor{softcolor}{HTML}{666666}    % Subdued Gray
\definecolor{hfcolor}{HTML}{677CA6}      % Header/Footer Slate Blue
\definecolor{hfrule}{HTML}{A5B4D6}       % Header/Footer Border Accent
\definecolor{codebg}{HTML}{F7F7F7}       % Code Container Fill
\definecolor{codeborder}{HTML}{D6D6D6}   % Code Container Stroke
\definecolor{dividerline}{HTML}{A0A0A0}  % Metadata Box Divider

% Thiết lập lề trang chuẩn A4 (2.5cm đều 4 cạnh giống Typst)
\usepackage{geometry}
\geometry{
    a4paper,
    margin=2.5cm,
    headheight=16pt,
    footskip=25pt
}

% Căn chỉnh đoạn văn bản
\usepackage{setspace}
\setstretch{1.15}
\setlength{\parindent}{1.2em}
\setlength{\parskip}{0.4em}

% Đồ họa và TikZ
\usepackage{graphicx}
\usepackage{tikz}
\usetikzlibrary{calc, positioning}
\usepackage{float}
\usepackage{placeins}

% Header và Footer
\usepackage{fancyhdr}
\usepackage{lastpage}

% Định dạng tiêu đề các mục (Giống 100% Typst heading - Mỗi mục lớn tự động qua trang mới)
\usepackage{titlesec}
\newcommand{\sectionbreak}{\clearpage}
\titleformat{\section}[block]
  {\normalfont\fontsize{18pt}{22pt}\bfseries\color{titlecolor}}
  {\thesection\hspace{0.4em}{\color{titlecolor!65!black}\rule[-0.15em]{1.5pt}{1.15em}}\hspace{0.4em}}
  {0pt}
  {}
\titlespacing*{\section}{0pt}{10pt}{14pt}

\titleformat{\subsection}[block]
  {\normalfont\fontsize{16pt}{20pt}\bfseries\color{titlecolor}}
  {\thesubsection\hspace{0.5em}}
  {0pt}
  {}
\titlespacing*{\subsection}{0pt}{12pt}{8pt}

\titleformat{\subsubsection}[block]
  {\normalfont\fontsize{13pt}{17pt}\bfseries\color{titlecolor}}
  {\thesubsubsection\hspace{0.5em}}
  {0pt}
  {}
\titlespacing*{\subsubsection}{0pt}{10pt}{6pt}

% Định dạng Mục lục (TOC)
\usepackage{tocloft}
\renewcommand{\contentsname}{\hfill\fontsize{18pt}{22pt}\selectfont\bfseries\color{titlecolor}Mục lục\hfill}
\renewcommand{\cftaftertoctitle}{\hfill}
\renewcommand{\cftsecfont}{\bfseries\color{titlecolor}}
\renewcommand{\cftsecpagefont}{\bfseries\color{titlecolor}}
\cftsetindents{section}{0em}{2em}
\cftsetindents{subsection}{2em}{2.2em}
\cftsetindents{subsubsection}{4.2em}{3em}
\setcounter{tocdepth}{3}

% Chú thích bảng biểu và hình ảnh
\usepackage{caption}
\captionsetup{font={small,color=softcolor},labelfont={bf,color=softcolor}}

% Định dạng khối mã nguồn (Code Block)
\usepackage{listings}
\lstset{
  basicstyle=\ttfamily\small\color{bodycolor},
  backgroundcolor=\color{codebg},
  frame=single,
  rulecolor=\color{codeborder},
  numbers=left,
  numberstyle=\tiny\color{softcolor},
  stepnumber=1,
  numbersep=8pt,
  tabsize=2,
  breaklines=true,
  breakatwhitespace=true,
  showspaces=false,
  showstringspaces=false
}

% Siêu liên kết (Hyperlinks)
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=titlecolor,
  filecolor=titlecolor,
  citecolor=titlecolor,
  urlcolor=titlecolor
}

% Trích dẫn tài liệu tham khảo (BibLaTeX)
\usepackage{csquotes}
\usepackage[backend=biber,style=numeric,sorting=none]{biblatex}
\addbibresource{ref.bib}
